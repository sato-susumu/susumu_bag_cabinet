# クラス使用例

このドキュメントでは、新しいMVCクラス構造の使い方を示します。

## Model の使用例

### BagFile モデルの使用

```python
from susumu_bag_cabinet.models.bag_file import BagFile
from pathlib import Path

# BagFileオブジェクトを作成
bag_file = BagFile("/home/user/bags/my_bag")

# プロパティにアクセス
print(f"ファイル名: {bag_file.name}")
print(f"パス: {bag_file.path}")

# bag情報を更新
info = {
    "size": 1024000,
    "format": "MCAP",
    "compression": "圧縮(Zstd)",
    "is_valid": "OK"
}
bag_file.update_info(info)

# 相対パスを取得
base_folder = Path("/home/user/bags")
relative = bag_file.get_relative_path(base_folder)
print(f"相対パス: {relative}")  # "my_bag"

# 整合性エラーをチェック
if bag_file.has_integrity_error():
    print("このファイルにはエラーがあります")

# シグナルに接続
def on_data_updated():
    print("データが更新されました")

bag_file.data_updated.connect(on_data_updated)
```

### BagCollection モデルの使用

```python
from susumu_bag_cabinet.models.bag_collection import BagCollection

# コレクションを作成
collection = BagCollection("/home/user/bags")

# シグナルに接続
def on_file_added(bag_file):
    print(f"追加: {bag_file.name}")

def on_collection_updated():
    print(f"合計: {collection.count()}個のファイル")

collection.file_added.connect(on_file_added)
collection.collection_updated.connect(on_collection_updated)

# ファイルを追加
bag_file = collection.add_file("/home/user/bags/my_bag")

# すべてのファイルを取得
for bag_file in collection.files:
    print(bag_file.name)

# パスから取得
bag_file = collection.get_file("/home/user/bags/my_bag")

# エラーのあるファイルを取得
error_files = collection.get_files_with_errors()
print(f"エラーファイル数: {len(error_files)}")

# フォルダをスキャン（注意: 同期処理、UIスレッドでは使用しない）
# collection.scan_folder()
```

### Recording モデルの使用

```python
from susumu_bag_cabinet.models.recording import Recording
from pathlib import Path

# Recordingオブジェクトを作成
recording = Recording()

# シグナルに接続
def on_status_updated(status):
    print(f"ステータス: {status}")

def on_elapsed_time_updated(seconds):
    print(f"経過時間: {seconds}秒")

def on_file_size_updated(size_bytes):
    print(f"ファイルサイズ: {size_bytes} バイト")

recording.status_updated.connect(on_status_updated)
recording.elapsed_time_updated.connect(on_elapsed_time_updated)
recording.file_size_updated.connect(on_file_size_updated)

# 記録を開始
output_path = Path("/home/user/bags/new_recording")
success = recording.start(output_path, label="テスト記録")

if success:
    print("記録を開始しました")

# 記録中かチェック
if recording.is_recording:
    print("記録中です")

# 経過時間を取得
elapsed = recording.get_elapsed_time()
print(f"経過時間: {elapsed}秒")

# 記録を停止
recording.stop()
```

## Controller の使用例

### BrowseController の使用

```python
from susumu_bag_cabinet.controllers.browse_controller import BrowseController
from susumu_bag_cabinet.models.bag_collection import BagCollection

# モデルを作成
collection = BagCollection("/home/user/bags")

# コントローラーを作成
controller = BrowseController(collection)

# シグナルに接続
def on_operation_started(operation):
    print(f"操作開始: {operation}")

def on_operation_progress(message):
    print(f"進捗: {message}")

def on_operation_completed(result):
    print(f"完了: {result}")

controller.operation_started.connect(on_operation_started)
controller.operation_progress.connect(on_operation_progress)
controller.operation_completed.connect(on_operation_completed)

# スキャンを開始
controller.start_scan()

# ファイルを圧縮（非同期で実行することを推奨）
files_to_compress = collection.files[:3]  # 最初の3ファイル
controller.compress_files(files_to_compress, compression="zstd")

# エラーのあるファイルを修復
error_files = collection.get_files_with_errors()
if error_files:
    controller.repair_files(error_files)

# ファイルを削除
files_to_delete = [collection.files[0]]
controller.delete_files(files_to_delete)
```

## View での使用例

### BrowsePage でのController使用

```python
# browse_page.py の一部を抜粋

from PySide6.QtWidgets import QWidget
from susumu_bag_cabinet.controllers.browse_controller import BrowseController
from susumu_bag_cabinet.models.bag_collection import BagCollection

class BrowsePage(QWidget):
    def __init__(self, config):
        super().__init__()

        # モデルを作成
        self.collection = BagCollection(config.get_bag_folder())

        # コントローラーを作成
        self.controller = BrowseController(self.collection)

        # シグナルを接続
        self.controller.operation_started.connect(self._on_operation_started)
        self.controller.operation_progress.connect(self._on_operation_progress)
        self.controller.operation_completed.connect(self._on_operation_completed)

        self.collection.file_added.connect(self._on_file_added)
        self.collection.collection_updated.connect(self._on_collection_updated)

    def start_scan(self):
        """スキャンを開始"""
        self.controller.start_scan()

    def _on_file_added(self, bag_file):
        """ファイルが追加されたときの処理"""
        # テーブルに行を追加
        row = self.table.rowCount()
        self.table.insertRow(row)
        # ...

    def _on_operation_started(self, operation):
        """操作開始時の処理"""
        # プログレスダイアログを表示
        self.progress_dialog.show()

    def _on_operation_progress(self, message):
        """操作進捗時の処理"""
        # プログレスダイアログのメッセージを更新
        self.progress_dialog.setLabelText(message)

    def _on_operation_completed(self, result):
        """操作完了時の処理"""
        # プログレスダイアログを閉じる
        self.progress_dialog.close()
        # 結果を表示
        QMessageBox.information(self, "完了", result)

    def compress_selected(self):
        """選択されたファイルを圧縮"""
        selected_files = self._get_selected_files()
        self.controller.compress_files(selected_files)
```

### RecordPage でのController使用

```python
# record_page.py の一部を抜粋

from PySide6.QtWidgets import QWidget
from susumu_bag_cabinet.controllers.record_controller import RecordController

class RecordPage(QWidget):
    def __init__(self, config):
        super().__init__()

        # コントローラーを作成
        self.controller = RecordController(
            base_folder=config.get_bag_folder(),
            robot_name=config.get_robot_name(),
            include_robot=config.get_folder_include_robot_name()
        )

        # シグナルを接続
        self.controller.recording_started.connect(self._on_recording_started)
        self.controller.recording_stopped.connect(self._on_recording_stopped)
        self.controller.status_updated.connect(self._update_status_label)
        self.controller.elapsed_time_updated.connect(self._update_time_label)
        self.controller.file_size_updated.connect(self._update_size_label)

    def start_recording(self):
        """記録を開始"""
        label = self.label_input.text()
        success = self.controller.start_recording(label)

        if success:
            self.path_label.setText(self.controller.get_output_path())

    def stop_recording(self):
        """記録を停止"""
        self.controller.stop_recording()

    def _on_recording_started(self):
        """記録開始時の処理"""
        self.start_btn.setEnabled(False)
        self.stop_btn.setEnabled(True)

    def _on_recording_stopped(self):
        """記録停止時の処理"""
        self.start_btn.setEnabled(True)
        self.stop_btn.setEnabled(False)

    def _update_time_label(self, seconds):
        """時間ラベルを更新"""
        time_str = self.controller.format_elapsed_time()
        self.time_label.setText(time_str)

    def _update_size_label(self, size_bytes):
        """サイズラベルを更新"""
        size_str = self.controller.format_file_size(size_bytes)
        self.size_label.setText(size_str)
```

## まとめ

### クラス使用のベストプラクティス

1. **Model**:
   - データの保持と管理のみ
   - UI操作を含めない
   - シグナルでデータ変更を通知

2. **Controller**:
   - ビジネスロジックを実装
   - ModelとViewを仲介
   - 重い処理は非同期で実行

3. **View**:
   - UI表示のみ
   - Controllerを経由してModelにアクセス
   - シグナルでUI更新

この設計により、テスト、保守、拡張が容易になります。
